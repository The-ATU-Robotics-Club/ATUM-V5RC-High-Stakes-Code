20 UKF(const StateCovariance &iP,
21 const StateCovariance &iQ,
22 const OutputCovariance &iR,
23 const double alpha = 0.5,
24 const double beta = 2.0) :

26 Q{iQ},

28 const double alpha2{alpha * alpha};
29 const double lambda{TotalStates * (alpha2 - 1.0)};
30 Wm[0] = lambda / (TotalStates + lambda);
31 Wc[0] = Wm[0] + (1.0 - alpha2 + beta);
32 for(int i{0}; i < Wm.size(); i++) {
33 Wm[i] = Wc[i] = 1.0 / (2.0 * (TotalStates + lambda));
34 }
35 eta = std::sqrt(TotalStates + lambda);
36 }

40 }

46 void predict() {

48
49 StateCovariance PSqrt{P.sqrt()};
50 SigmaPoints chi;
51 chi << xHat, merweAdd(eta * PSqrt, xHat), merweAdd(-eta * PSqrt, xHat);
52
53 chiF = chi;
54 for(auto point : chiF.colwise()) {
55 point = f(point, u);
56 }
57
58 xHatPriori = State{};
59 for(int i{0}; i < chiF.size(); i++) {
60 xHatPriori += Wm[i] * chiF.col(i);
61 }
62
63 PPriori = Q;
64 for(int i{0}; i < Wc.size(); i++) {
65 const State diff{chiF.col(i) - xHatPriori};
66 PPriori += Wc[i] * diff * diff.transpose();
67 }
68
69 upsilon = SigmaOutputPoints{};
70 for(int i{0}; i < chiF.cols(); i++) {
71 upsilon.col(i) = h(chiF.col(i), u);
72 }
73
74 yHatPriori = Output{};
75 for(int i{0}; i < upsilon.size(); i++) {
76 yHatPriori += Wm[i] * upsilon.col(i);
77 }
78 }

80 void update() {
81 const Output y{getOutput()};
82
83 OutputCovariance Py{R};
84 for(int i{0}; i < Wc.size(); i++) {
85 const Output diff{upsilon.col(i) - yHatPriori};
86 Py += Wc[i] * diff * diff.transpose();
87 }
88
89 Eigen::Matrix<double, TotalStates, TotalOutputs> Pxy;
90 for(int i{0}; i < Wc.size(); i++) {
91 const State xDiff{chiF.col(i) - xHatPriori};
92 const Output yDiff{upsilon.col(i) - yHatPriori};
93 Pxy += Wc[i] * xDiff * yDiff.transpose();
94 }
95
96 const KalmanGain K{Pxy * Py.inverse()};

98 xHat = xHatPriori + K * (y - yHatPriori);
99 P = PPriori - K * Py * K.transpose();
